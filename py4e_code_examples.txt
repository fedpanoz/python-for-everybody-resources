# CAPITOLO 1: Introduzione (words.py)
# Contare la parola più frequente in un file di testo
# Fonte: http://www.py4e.com/code3/words.py
# ------------------------------------------------------------------------------
name = input('Enter file:')
handle = open(name, 'r')
counts = dict()

for line in handle:
    words = line.split()
    for word in words:
        counts[word] = counts.get(word, 0) + 1

bigcount = None
bigword = None
for word, count in list(counts.items()):
    if bigcount is None or count > bigcount:
        bigword = word
        bigcount = count

print(bigword, bigcount)

# CAPITOLO 10: Tuple e ordinamento di dizionari per valore (count3.py)
# Rimozione punteggiatura con translate() e top 10 parole più frequenti
# Fonte: http://www.py4e.com/code3/count3.py
# ------------------------------------------------------------------------------
import string

fhand = open('romeo-full.txt')
counts = dict()
for line in fhand:
    line = line.translate(str.maketrans('', '', string.punctuation))
    line = line.lower()
    words = line.split()
    for word in words:
        if word not in counts:
            counts[word] = 1
        else:
            counts[word] += 1

# Sort the dictionary by value
lst = list()
for key, val in list(counts.items()):
    lst.append((val, key))

lst.sort(reverse=True)

for key, val in lst[:10]:
    print(key, val)

# Esercizio 2 Cap. 11: Estrarre numeri da "New Revision: ..." e calcolare la media
fname = input('Enter file: ')
try:
    fhand = open(fname)
except:
    print('File cannot be opened:', fname)
    exit()

numeri = list()
for line in fhand:
    line = line.rstrip()
    x = re.findall(r'^New Revision: ([0-9]+)', line)
    if len(x) > 0:
        valore = float(x[0])
        numeri.append(valore)

if len(numeri) > 0:
    media = sum(numeri) / len(numeri)
    print(int(media))

# CAPITOLO 12: Programmi di Rete - Il browser web più semplice del mondo (socket1.py)

import socket

mysock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
mysock.connect(('data.pr4e.org', 80))
cmd = 'GET http://data.pr4e.org/romeo.txt HTTP/1.0

'.encode()
mysock.send(cmd)

    data = mysock.recv(512)
    if len(data) < 1:
        break
    print(data.decode(), end='')

mysock.close()

# CAPITOLO 12: Parsing HTML con BeautifulSoup (urllinks.py)
# Fonte: http://www.py4e.com/code3/urllinks.py
# ------------------------------------------------------------------------------
import urllib.request, urllib.parse, urllib.error
from bs4 import BeautifulSoup
import ssl

# Ignore SSL certificate errors
ctx = ssl.create_default_context()
ctx.check_hostname = False
ctx.verify_mode = ssl.CERT_NONE

url = input('Enter - ')
html = urllib.request.urlopen(url, context=ctx).read()
soup = BeautifulSoup(html, 'html.parser')

# Retrieve all of the anchor tags
tags = soup('a')
for tag in tags:
    print(tag.get('href', None))

# CAPITOLO 13: Web Services e JSON (frammento interrogazione API / geocoding)
# ------------------------------------------------------------------------------
import urllib.request, urllib.parse, urllib.error
import json
import ssl

api_key = False
serviceurl = 'http://py4e-data.dr-chuck.net/json?'

# Ignore SSL certificate errors
ctx = ssl.create_default_context()
ctx.check_hostname = False
ctx.verify_mode = ssl.CERT_NONE

while True:
    address = input('Enter location: ')
    if len(address) < 1:
        break

    parms = dict()
    parms['address'] = address
    if api_key is not False:
        parms['key'] = api_key
    url = serviceurl + urllib.parse.urlencode(parms)

    print('Retrieving', url)
    uh = urllib.request.urlopen(url, context=ctx)
    data = uh.read().decode()
    print('Retrieved', len(data), 'characters')

    try:
        js = json.loads(data)
    except:
        js = None
